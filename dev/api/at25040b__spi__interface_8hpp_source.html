201constexpr uint8_t AddressedOpcode(uint8_t opcode, uint16_t address) noexcept {
202 return ((static_cast<uint32_t>(address) & static_cast<uint32_t>(reg::kAddressA8)) != 0U)
203 ? static_cast<uint8_t>(static_cast<uint32_t>(opcode) |
204 static_cast<uint32_t>(reg::kOpcodeA8))
205 : opcode;
206}

209constexpr uint8_t AddressByte(uint16_t address) noexcept {
210 return static_cast<uint8_t>(static_cast<uint32_t>(address) &
211 static_cast<uint32_t>(reg::kAddressLowMask));
212}

286
289 Error Read(uint16_t address, std::span<uint8_t> out) noexcept {
290 if (out.empty() || (out.size() > kReadChunkBytes) || (address > reg::kAddressMax)) {
291 return Error::InvalidArgument;
292 }
293 tx_.fill(0x00U);
294 // MISO idles high: a port that reports success without writing `rx` yields 0xFF, never the
295 // bytes of an earlier transfer (KB-CPP-116).
296 rx_.fill(reg::kStatusBusyByte);
297 tx_[0] = AddressedOpcode(reg::kRead, address);
298 tx_[1] = AddressByte(address);
299 if (!port_.Transfer(tx_.data(), rx_.data(), kCommandBytes + out.size())) {
300 return Error::Bus;
301 }
302 const std::size_t count = out.size();
303 for (std::size_t i = 0U; i < count; ++i) {
304 out[i] = rx_[kCommandBytes + i];
305 }
306 return Error::Ok;
307 }

310 template <std::size_t N>
311 requires WriteSize<N>
312 Error Write(uint16_t address, std::span<const uint8_t, N> data) noexcept {
313 return Write(address, std::span<const uint8_t>{data});
314 }

317 template <std::size_t N>
318 requires((N != std::dynamic_extent) && !WriteSize<N>)
319 Error Write(uint16_t address, std::span<const uint8_t, N> data) noexcept = delete;
320
325 Error Write(uint16_t address, std::span<const uint8_t> data) noexcept {
326 const std::size_t offset = static_cast<std::size_t>(address) % kPageSizeBytes;
327 if (data.empty() || (data.size() > (kPageSizeBytes - offset)) || (address > reg::kAddressMax)) {
328 return Error::InvalidArgument;
329 }
330 tx_[0] = AddressedOpcode(reg::kWrite, address);
331 tx_[1] = AddressByte(address);
332 const std::size_t count = data.size();
333 for (std::size_t i = 0U; i < count; ++i) {
334 tx_[kCommandBytes + i] = data[i];
335 }
336 return port_.Transfer(tx_.data(), nullptr, kCommandBytes + data.size()) ? Error::Ok
337 : Error::Bus;
338 }
