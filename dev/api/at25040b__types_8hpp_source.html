307constexpr std::size_t ProtectedFrom(Variant variant, BlockProtect level) noexcept {
308 const std::size_t size = CapacityBytes(variant);
309 switch (level) {
310 case BlockProtect::None:
311 return size;
312 case BlockProtect::UpperQuarter:
313 return size - (size / 4U);
314 case BlockProtect::UpperHalf:
315 return size / 2U;
316 case BlockProtect::All:
317 return 0U;
318 default:
319 break;
320 }
321 return 0U;
322}

324// Table 6-4, the AT25040B column: 180-1FF, 100-1FF, 000-1FF protected.
325static_assert(ProtectedFrom(Variant::At25040b, BlockProtect::UpperQuarter) == 0x180U);
326static_assert(ProtectedFrom(Variant::At25040b, BlockProtect::UpperHalf) == 0x100U);
327static_assert(ProtectedFrom(Variant::At25010b, BlockProtect::UpperQuarter) == 0x60U);
328static_assert(ProtectedFrom(Variant::At25020b, BlockProtect::UpperHalf) == 0x80U);

380constexpr Error Validate(const Config& config) noexcept {
381 if (!detail::IsEnumerator(config.variant)) {
382 return Error::InvalidArgument;
383 }
384 if ((config.write_timeout_us < kWriteCycleMaxUs) ||
385 (config.write_timeout_us > kWriteTimeoutMaxUs)) {
386 return Error::InvalidArgument;
387 }
388 return Error::Ok;
389}
