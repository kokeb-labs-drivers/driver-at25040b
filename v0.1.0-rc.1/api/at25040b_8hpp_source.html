34// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept the exact
35// signature, `noexcept` and the result type; the driver rejects a hook that is declared but does
36// not match (KB-CPP20-007).

39template <typename Port>
40concept DeclaresGpioSet =
41 requires(Port& port) { port.GpioSet(CtrlPin::WriteProtect, GpioSignal::ACTIVE); };
42
44template <typename Port>
45concept HasGpioSet =
46 requires { ExactHook<Port, bool(CtrlPin, GpioSignal) noexcept>::Take(&Port::GpioSet); };

49template <typename Port>
50concept DeclaresDelayUs = requires(Port& port) { port.DelayUs(1U); };

54template <typename Port>
55concept HasDelayUs = requires { ExactHook<Port, void(uint32_t) noexcept>::Take(&Port::DelayUs); };

59concept DeclaresLog = requires(Port& port) { port.Log(LogLevel::Info, ""); };

106constexpr Error ToggleError(uint8_t got, uint8_t expected) noexcept {
107 const bool only_wen =
108 reg::IsIdleStatus(got) &&
109 ((static_cast<uint32_t>(got) ^ static_cast<uint32_t>(expected)) == reg::kStatusWen);
110 return only_wen ? Error::DeviceIdMismatch : Error::BusNoise;
111}
112
113static_assert((DecodeBlockProtect(0x0C_u8) == BlockProtect::All) &&
114 (DecodeBlockProtect(0x06_u8) == BlockProtect::UpperQuarter));
115static_assert((EncodeBlockProtect(BlockProtect::UpperHalf) == reg::kStatusBp1) &&
116 (DecodeBlockProtect(EncodeBlockProtect(BlockProtect::UpperQuarter)) ==
117 BlockProtect::UpperQuarter));
118static_assert((ToggleError(0x00_u8, reg::kStatusWen) == Error::DeviceIdMismatch) &&
119 (ToggleError(0x04_u8, reg::kStatusWen) == Error::BusNoise) &&
120 (ToggleError(reg::kStatusBusyByte, 0x00_u8) == Error::BusNoise));
